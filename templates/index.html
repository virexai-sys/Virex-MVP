/* ======================================================== CHAT — REAL FLASK API ======================================================== */ async function handleChatSubmit(e) { e.preventDefault(); const input = document.getElementById('chat-input'); const msg = input.value.trim(); if (!msg) return; const chatMessages = document.getElementById('chat-messages'); /* USER MESSAGE */ const userMessage = document.createElement('div'); userMessage.className = 'flex items-start justify-end gap-3'; userMessage.innerHTML = `
${escapeHtml(msg)}
U
`; chatMessages.appendChild(userMessage); input.value = ''; chatMessages.scrollTop = chatMessages.scrollHeight; /* TYPING INDICATOR */ const typingMessage = document.createElement('div'); typingMessage.id = 'ai-typing'; typingMessage.className = 'flex items-start gap-3'; typingMessage.innerHTML = `
EZKROY AI is thinking...
`; chatMessages.appendChild(typingMessage); chatMessages.scrollTop = chatMessages.scrollHeight; /* REAL BACKEND REQUEST */ try { const response = await fetch('/api/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: msg }) }); if (!response.ok) { throw new Error( `Server error: ${response.status}` ); } const data = await response.json(); /* REMOVE TYPING */ const typing = document.getElementById('ai-typing'); if (typing) { typing.remove(); } /* AI RESPONSE */ const botMessage = document.createElement('div'); botMessage.className = 'flex items-start gap-3'; botMessage.innerHTML = `
${formatAIResponse( data.reply || 'দুঃখিত, কোনো উত্তর পাওয়া যায়নি।' )}
`; chatMessages.appendChild(botMessage); /* ORDER CREATED */ if (data.order_created && data.order) { const orderNotice = document.createElement('div'); orderNotice.className = 'ml-12 text-xs text-emerald-600 font-semibold mt-1'; orderNotice.innerHTML = ` Order #${data.order.id} saved successfully `; chatMessages.appendChild(orderNotice); } chatMessages.scrollTop = chatMessages.scrollHeight; } catch (error) { console.error( 'CHAT API ERROR:', error ); const typing = document.getElementById('ai-typing'); if (typing) { typing.remove(); } const errorMessage = document.createElement('div'); errorMessage.className = 'flex items-start gap-3'; errorMessage.innerHTML = `
AI server-এর সাথে যোগাযোগ করা যাচ্ছে না। Flask server এবং OPENAI_API_KEY check করুন।
`; chatMessages.appendChild(errorMessage); chatMessages.scrollTop = chatMessages.scrollHeight; } } /* ======================================================== SECURITY — HTML ESCAPE ======================================================== */ function escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; } /* ======================================================== AI RESPONSE FORMATTER ======================================================== */ function formatAIResponse(text) { let safeText = escapeHtml(text); safeText = safeText.replace( /\*\*(.*?)\*\*/g, '$1' ); safeText = safeText.replace( /\n/g, '
' ); return safeText; }
